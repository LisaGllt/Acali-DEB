3.1 Protocol

Synchronized newly hatched earthworms are put in cosms with 200 g of Closeaux soil with a WHC of 70%. They are fed every 14 days and the soil is renewed every 28 days. Soil humidity is checked every week. Earthworms are weighted every 14 days. Once at least one earthworm is adult, cocoons produced are retrieved every 28 days when changing the soil.

Conditions :

  • D1 : One earthworm per cosm | Food = 3 g/14 days
  • D2 : 2 earthworms per cosm | Food = 3 g/14 days
  • D2b : 2 earthworms per cosm | Food = 6 g/14 days
  • D5 : 5 earthworms per cosm | Food = 3 g/14 days
  • D7 : 7 earthworms per cosm | Food = 3 g/14 days

Some earthworm died in the first 14 days of the experiment. Another batch of earthworms were then started 28 days after the first one.

f_GrowthRate <- function(data){
 
  df_GrowthRate <- data.frame() 
 
  for (i in unique(data$ID_cosm)){

    data_i <- subset(data, ID_cosm == i) 
    Nb_vdt_i <- unique(subset(data_i, t==0)$Nb_vdt)
    Food_i <- unique(subset(data_i, t==0)$Food)
    Nb_rep_i <- unique(subset(data_i, t==0)$Nb_rep)
    Lot_i <- unique(subset(data_i, t==0)$Lot)
    Condition_i <- unique(subset(data_i, t==0)$Condition)
   
    if (length(na.omit(data_i$L))<2) {
      a.Est <- NA
      
    }else if(length(na.omit(data_i$L))==2){ 
     
      a.Est <- (subset(data_i, !t==0)$L - subset(data_i, t==0)$L)/subset(data_i, !t==0)$t # (L(t=t)-L(t=0))/t
     
    }else{
      lm_i <- lm(L~t, data=data_i)
      a.Est <- lm_i$coefficients[2]
    }
   
    df_GrowthRate_i <- data.frame(
      ID_cosm = i,
      Nb_vdt = Nb_vdt_i,
      Food = Food_i,
      Nb_rep = Nb_rep_i,
      Lot = Lot_i,
      Condition = Condition_i,
      Growth_rate = a.Est
    )
    
    df_GrowthRate <- rbind(df_GrowthRate, df_GrowthRate_i)
  }
 
  return(df_GrowthRate) 
}

```{r growthrate, message=F, warning=F}
f_GrowthRate <- function(data){
 
  df_GrowthRate <- data.frame() 
 
  for (i in unique(data$ID_cosm)){

    data_i <- subset(data, ID_cosm == i) 
    Nb_vdt_i <- unique(subset(data_i, t==0)$Nb_vdt)
    Food_i <- unique(subset(data_i, t==0)$Food)
    Nb_rep_i <- unique(subset(data_i, t==0)$Nb_rep)
    Lot_i <- unique(subset(data_i, t==0)$Lot)
    Condition_i <- unique(subset(data_i, t==0)$Condition)
   
    if (length(na.omit(data_i$L))<2) {
      a.Est <- NA
      
    }else if(length(na.omit(data_i$L))==2){ 
     
      a.Est <- (subset(data_i, !t==0)$L - subset(data_i, t==0)$L)/subset(data_i, !t==0)$t # (L(t=t)-L(t=0))/t
     
    }else{
      lm_i <- lm(L~t, data=data_i)
      a.Est <- lm_i$coefficients[2]
    }
   
    df_GrowthRate_i <- data.frame(
      ID_cosm = i,
      Nb_vdt = Nb_vdt_i,
      Food = Food_i,
      Nb_rep = Nb_rep_i,
      Lot = Lot_i,
      Condition = Condition_i,
      Growth_rate = a.Est
    )
    
    df_GrowthRate <- rbind(df_GrowthRate, df_GrowthRate_i)
  }
 
  return(df_GrowthRate) 
}
```

```{r}
df_dens_repro_clean <- df_dens_repro |>

  mutate(
    start_repro = min(t[Nb_cocoons > 0], na.rm = TRUE)
  ) |>
  ungroup() |>
  filter(is.finite(start_repro)) |>
  arrange(ID, t) |>
  group_by(ID) |>
  mutate(
    dt = t - lag(t)
  ) |>
  ungroup()  |>
  mutate(

  filter(Repro_rate!=0) |> 
  group_by(ID) |> 
  slice(-1) |> # Remove first repro rate bc reproduction started some unknow time during the previous 28 days.
  summarise(
    Repro_rate = mean(Repro_rate, na.rm = TRUE),
    Condition = first(Condition),
    Food = first(Food),
    Nb_vdt = first(Nb_vdt),
    .groups = "drop"
  ) |> 
  ungroup()
```
